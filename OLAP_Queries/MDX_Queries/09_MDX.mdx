    [Measures].[Gross Approval] ON COLUMNS,

    TOPCOUNT(
        NONEMPTY(
            [Dim Project Geography].[Project County].[Project County].Members,
            [Measures].[Gross Approval]
        ),
        3,
        [Measures].[Gross Approval]
    ) ON ROWS

    [Dim Project Geography].[Project State].&[CA]
);
