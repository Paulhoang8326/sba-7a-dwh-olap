SELECT
    {
        [Measures].[Gross Charge Off Amount],
        [Measures].[Record Count]
    } ON COLUMNS,

    TOPCOUNT(
        [Dim Industry].[Naics Sector Code].[Naics Sector Code].MEMBERS,
        5,
        [Measures].[Gross Charge Off Amount]
    ) ON ROWS

FROM [SBA7A DWH_DSV]

WHERE (
    [Dim Date].[Fiscal Year].&[2023],
    [Dim Loan Status].[Raw Status].&[CHGOFF]
);
